📟 This MikroC PRO for ARM code runs on a microcontroller connected to an ADXL345 accelerometer via I2C. It reads vibration data, computes RMS, and sends it via UART to the BearingIQ server.
// BearingIQ — Embedded Sensor Acquisition
// Platform: STM32F4 | MikroC PRO for ARM v7.x
// Sensor  : ADXL345 Accelerometer (I2C)

#define SAMPLE_RATE   48000   // Hz - matches CWRU dataset
#define BUFFER_SIZE   1024    // samples per window
#define ADXL345_ADDR  0x53    // I2C slave address
#define DATA_FORMAT   0x31    // +/- 16g range, full res

float vibBuffer[BUFFER_SIZE];
float features[9];  // max, min, mean, sd, rms, skew, kurt, crest, form

// ── Initialize ADXL345 accelerometer ─────────────────
void ADXL345_Init() {
    I2C1_Start();
    I2C1_Write(0x2D, 0x08);  // Power CTL: measure mode
    I2C1_Write(DATA_FORMAT, 0x0B);  // Full resolution, ±16g
    I2C1_Write(0x2C, 0x0F);  // BW_RATE: 3200 Hz ODR
    I2C1_Stop();
}

// ── Read one Z-axis vibration sample ─────────────────
float ReadAccelZ() {
    uint8_t data[2];
    int16_t raw;
    I2C1_Start();
    I2C1_Write(ADXL345_ADDR);
    I2C1_Write(0x36);  // DATAZ0 register
    I2C1_Repeated_Start();
    I2C1_Read(ADXL345_ADDR, data, 2, END_MODE_STOP);
    raw = (data[1] << 8) | data[0];
    return raw * 0.004f;  // Convert to g (4mg/LSB)
}

// ── Compute time-domain features ─────────────────────
void ComputeFeatures(float* buf, int n) {
    float sum=0, sumSq=0, maxV=-9999, minV=9999;
    for(int i=0; i<n; i++) {
        sum   += buf[i];
        sumSq += buf[i] * buf[i];
        if(buf[i] > maxV) maxV = buf[i];
        if(buf[i] < minV) minV = buf[i];
    }
    float mean = sum / n;
    float rms  = sqrt(sumSq / n);
    float sd   = sqrt((sumSq/n) - mean*mean);

    // Skewness and kurtosis (3rd & 4th moments)
    float m3=0, m4=0;
    for(int i=0; i<n; i++) {
        float d = buf[i] - mean;
        m3 += d*d*d;
        m4 += d*d*d*d;
    }
    float skew = (m3/n) / (sd*sd*sd);
    float kurt = (m4/n) / (sd*sd*sd*sd) - 3.0f;

    features[0] = maxV;           // max
    features[1] = minV;           // min
    features[2] = mean;           // mean
    features[3] = sd;             // sd
    features[4] = rms;            // rms
    features[5] = skew;           // skewness
    features[6] = kurt;           // kurtosis
    features[7] = maxV / rms;     // crest factor
    features[8] = rms / mean;     // form factor
}

// ── Send JSON to BearingIQ API via UART/WiFi ──────────
void SendToAPI() {
    char json[256];
    sprintf(json,
        "{\"features\":[%.4f,%.4f,%.4f,%.4f,%.4f,%.4f,%.4f,%.4f,%.4f],\"model\":\"random_forest\"}",
        features[0], features[1], features[2],
        features[3], features[4], features[5],
        features[6], features[7], features[8]
    );
    UART1_Write_Text(json);  // Send via WiFi module
}

// ── Main loop ─────────────────────────────────────────
void main() {
    ADXL345_Init();
    while(1) {
        // Acquire 1024 samples
        for(int i=0; i<BUFFER_SIZE; i++) {
            vibBuffer[i] = ReadAccelZ();
            Delay_us(21);  // ~48 kHz sampling rate
        }
        ComputeFeatures(vibBuffer, BUFFER_SIZE);
        SendToAPI();
        Delay_ms(1000);  // Send every 1 second
    }
}
⚡ How it connects to BearingIQ: The microcontroller sends JSON to a WiFi module (ESP8266/ESP32). The HTTP POST is directed to http://<server-ip>:5000/api/predict. The server returns the fault class in <50ms, which the embedded system can display on an LCD or trigger an alarm relay.
🐍 This Python code loads the CWRU CSV data, computes features, and performs batch prediction — useful for offline analysis.
import pandas as pd
import numpy as np
import requests
from scipy import stats

# Load raw vibration CSV
df = pd.read_csv('archive/Bearing.csv')

# Features already extracted in the dataset
FEATURES = ['max', 'min', 'mean', 'sd', 'rms',
            'skewness', 'kurtosis', 'crest', 'form']

# Example: predict a single row
row = df.iloc[0]
features = [float(row[f]) for f in FEATURES]

# Call BearingIQ REST API
resp = requests.post(
    'http://localhost:5000/api/predict',
    json={'features': features, 'model': 'random_forest'}
)
result = resp.json()
print(f"Predicted: {result['label']} ({result['confidence']*100:.1f}% confidence)")
print(f"Severity : {result['severity']}/3")
print(f"Fault Type: {result['fault_type']}")
